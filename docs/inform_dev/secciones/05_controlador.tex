\section{Controlador}

El controlador es un modelo atómico DEVS cuyas variables de estado han sido divididas para gestionar de forma independiente el monitoreo del flujo y el ciclo de vida de la bolsa de infusión. Recibe órdenes médicas, lecturas del sensor, señales de fin de bolsa y confirmaciones del enfermero; emite ajustes a la bomba, registra eventos y dispara alarmas sin que una condición crítica bloquee a la otra.

El mecanismo de salida se basa en una \textbf{cola de acciones pendientes} $\mathit{acciones} \in (A^* \times \mathbb{R}_{\ge 0})^*$.

\subsection{Definición formal}

\paragraph{Puertos.}
\begin{align*}
  X &= \{\mathrm{ordenMedica}\}\!\times\!\{0\}
        \;\cup\; \{\mathrm{sensorFlujo}\}\!\times\!\{1\} \\
    &\quad \cup\; \{\mathrm{finBolsa}\}\!\times\!\{2\}
        \;\cup\; \{\mathrm{confirmacionEnfermero}\}\!\times\!\{3\}
  \\[4pt]
  Y &= \{\mathrm{alarmaBaja},\mathrm{alarmaMedia},\mathrm{alarmaCritica}\}
            \!\times\!\{0\} \\
    &\quad \cup\;
        \{\mathrm{ajustarCaudal},\mathrm{detenerBomba},\mathrm{registrarEvento}\}
            \!\times\!\{1\} \\
    &\quad \cup\;
        \{\mathrm{registrarEvento}\}\!\times\!\{9\}
\end{align*}

\paragraph{Espacio de estados.}
\begin{align*}
  S &= \underbrace{\mathbb{R}_{\ge 0}}_{\mathit{uCM}}
     \times
     \underbrace{\mathbb{R}_{\ge 0}}_{\mathit{cO}}
     \times
     \underbrace{\mathbb{N}_0}_{\mathit{tol}}
     \times
     \underbrace{E_{\mathrm{flujo}}}_{\mathit{est}_{\mathrm{flujo}}} \\
    &\quad \times
     \underbrace{E_{\mathrm{bolsa}}}_{\mathit{est}_{\mathrm{bolsa}}}
     \times
     \underbrace{\bigl(\mathbb{R}_0^+ \cup \{\infty\}\bigr)}_{\tau_{\mathrm{bolsa}}}
     \times
     \underbrace{(A^* \times \mathbb{R}_{\ge 0})^*}_{\mathit{acciones}}
\end{align*}

donde los dominios discretos se definen como:
\begin{align*}
  E_{\mathrm{flujo}} &= \{\mathrm{normal},\,\mathrm{estadoMedio},\,\mathrm{estadoCritico}\} \\
  E_{\mathrm{bolsa}} &= \{\mathrm{llena},\,\mathrm{bolsaBaja},\,\mathrm{esperandoDetencion},\,\mathrm{vacia}\}
\end{align*}

y el conjunto de acciones etiquetadas posibles es:
\begin{align*}
  A^* &= \{\mathrm{alarmaBaja},\,\mathrm{alarmaMedia},\,\mathrm{alarmaCritica},\,
        \mathrm{detenerBomba},\,\mathrm{registrarEvento}\} \\
      &\quad \cup\; \{(\mathrm{ajustarCaudal},\, \Delta) \mid \Delta \in \mathbb{R}\} \\
      &\quad \cup\; \{(\mathrm{registrarEvento},\, s) \mid s \in S\}
\end{align*}

\paragraph{Roles funcionales de cada variable.}

\begin{table}[H]
\centering
\begin{tabular}{lll}
\toprule
Variable & Tipo & Rol \\
\midrule
$uCM$                   & $\mathbb{R}_{\ge 0}$               & Monitoreo de flujo \\
$cO$                    & $\mathbb{R}_{\ge 0}$               & Gestión de órdenes \\
$tol$                   & $\mathbb{N}_0$                     & Monitoreo de flujo \\
$est_{\mathrm{flujo}}$  & $E_{\mathrm{flujo}}$               & Control de anomalías de caudal \\
$est_{\mathrm{bolsa}}$  & $E_{\mathrm{bolsa}}$               & Ciclo de vida de la medicación \\
$\tau_{\mathrm{bolsa}}$ & $\mathbb{R}_0^+\cup\{\infty\}$     & Temporizador exclusivo de la bolsa \\
$\mathit{acciones}$     & $(A^* \times \mathbb{R}_{\ge 0})^*$ & Cola de salidas pendientes con demora \\
\bottomrule
\end{tabular}
\end{table}

\paragraph{Estado inicial.}
\[
  s_0 = (0,\;0,\;0,\;\mathrm{normal},\;\mathrm{llena},\;\infty,\;[])
\]

\paragraph{Función auxiliar \texttt{no\_tolerable}.}
\[
  \mathrm{no\_tolerable}(cO,uCM)
  \;=\;
  \frac{|cO-uCM|}{cO} > 0{,}10
\]

\paragraph{Función auxiliar \texttt{conLog}.}
\begin{align*}
  \mathrm{conLog}(s,\, []) &= [] \\[4pt]
  \mathrm{conLog}(s,\, [(a_1, d_1),\, \ldots]) &=
    [(a_1,\, d_1),\, ((\mathrm{registrarEvento},\, s),\, 0)]
    \;\mathbin{+\!\!+}\;
    \mathrm{conLog}(s,\, [(a_2, d_2), \ldots])
\end{align*}

\paragraph{Avance temporal.}
\[
  \ta(s) =
  \begin{cases}
    d_1
      & \text{si } \mathit{acciones} = [(a_1,d_1),\ldots] \\[4pt]
    \tau_{\mathrm{bolsa}}
      & \text{si } \mathit{acciones} = []
        \;\land\; est_{\mathrm{bolsa}} = \mathrm{esperandoDetencion} \\[4pt]
    \infty
      & \text{si } \mathit{acciones} = []
        \;\land\; est_{\mathrm{bolsa}} \neq \mathrm{esperandoDetencion}
  \end{cases}
\]

\paragraph{Caso de absorción global.}
\[
\delta_{\mathrm{ext}}(s,\,e,\,((c,t),0)) =
\begin{cases}
  \begin{aligned}
    \bigl(&uCM,\;c,\;0,\;\mathrm{normal},\;\mathrm{llena},\;\infty,\; \\
          &\mathrm{conLog}(s,\,[((\mathrm{ajustarCaudal},\,c - uCM),\; \mathrm{random}(0,3))])\bigr)
  \end{aligned}
  & \text{si } est_{\mathrm{bolsa}} = \mathrm{vacia} \land c > 0 \\[3ex]
  \begin{aligned}
    \bigl(&uCM,\;0,\;tol,\;est_{\mathrm{flujo}},\;est_{\mathrm{bolsa}},\; \tau_{\mathrm{bolsa}}-e,\; \\
          &\mathrm{conLog}(s,\,[(\mathrm{detenerBomba},\,0)])\bigr)
  \end{aligned}
  & \text{si } est_{\mathrm{bolsa}} = \mathrm{vacia} \land c = 0
\end{cases}
\]

\paragraph{Rol 1 – Gestión de órdenes.}

Transición externa — orden médica en estado crítico ($est_{\mathrm{flujo}} = \mathrm{estadoCritico}$):
\begin{align*}
  \delta_{\mathrm{ext}}(s,\,e,\,((c,t),0))
  &= \bigl(uCM,\;c,\;tol,\;\mathrm{estadoCritico},\;est_{\mathrm{bolsa}},\;
           \tau_{\mathrm{bolsa}}-e,\; []\bigr)
\end{align*}

Transición externa — orden médica con $c > 0$:
\begin{align*}
  \delta_{\mathrm{ext}}(s,\,e,\,((c,t),0))
  &= \bigl(uCM,\;c,\;tol,\;est_{\mathrm{flujo}},\;est_{\mathrm{bolsa}},\;
           \tau_{\mathrm{bolsa}}-e, \\
  &\qquad \mathrm{conLog}(s,\,[((\mathrm{ajustarCaudal},\, c - uCM),\;
             \mathrm{random\_truncado}(0.3) \times 10)])\bigr)
\end{align*}

Transición externa — orden médica con $c = 0$:
\begin{align*}
  \delta_{\mathrm{ext}}(s,\,e,\,((0,t),0))
  &= \bigl(uCM,\;0,\;tol,\;est_{\mathrm{flujo}},\;est_{\mathrm{bolsa}},\;
           \tau_{\mathrm{bolsa}}-e,\;
           \mathrm{conLog}(s,\,[(\mathrm{detenerBomba},\;0)])\bigr)
\end{align*}

\paragraph{Rol 2 – Monitoreo de flujo.}
Sea $\tau' = \tau_{\mathrm{bolsa}} - e$:
\[
  \delta_{\mathrm{ext}}(s,\,e,\,(x,1)) =
  \begin{cases}
    \begin{aligned}
      \bigl(&x,\,cO,\,0,\,\mathrm{estadoCritico},\,est_{\mathrm{bolsa}},\,\tau',\, \\
            &\mathrm{conLog}(s,\,[(\mathrm{alarmaCritica},0),\,(\mathrm{detenerBomba},0)])\bigr)
    \end{aligned}
    & \text{si } tol \ge 5 \land est_{\mathrm{flujo}} = \mathrm{estadoMedio}
    \\[3ex]
    \begin{aligned}
      \bigl(&x,\,cO,\,0,\,\mathrm{estadoMedio},\,est_{\mathrm{bolsa}},\,\tau',\, \\
            &\mathrm{conLog}(s,\,[(\mathrm{alarmaMedia},0)])\bigr)
    \end{aligned}
    & \text{si } tol \ge 5 \land est_{\mathrm{flujo}} = \mathrm{normal}
    \\[3ex]
    \begin{aligned}
      \bigl(&x,\,cO,\,tol+1,\,est_{\mathrm{flujo}},\,est_{\mathrm{bolsa}},\,\tau',\, \\
            &\mathrm{conLog}(s,\,[((\mathrm{ajustarCaudal},\,cO - x),\,0), \\
            &\qquad\qquad ((\mathrm{registrarEvento},\,\mathrm{snapshot}),\,0)])\bigr)
    \end{aligned}
    & \text{si } \mathrm{no\_tolerable}(cO,\,x) \land tol < 5
    \\[3ex]
    \begin{aligned}
      \bigl(&x,\,cO,\,0,\,est_{\mathrm{flujo}},\,est_{\mathrm{bolsa}},\,\tau',\, \\
            &\mathrm{conLog}(s,\,[((\mathrm{ajustarCaudal},\,cO - x),\,0), \\
            &\qquad\qquad ((\mathrm{registrarEvento},\,\mathrm{snapshot}),\,0)])\bigr)
    \end{aligned}
    & \text{en otro caso}
  \end{cases}
\]

Transición externa — confirmación del enfermero:
\[
  \delta_{\mathrm{ext}}(s,\,e,\,(\mathrm{confirmacionEnfermero},3)) =
  \begin{cases}
    \begin{aligned}
      \bigl(&uCM,\;cO,\;0,\;\mathrm{normal},\;\mathrm{llena},\;\infty,\; \\
            &\mathrm{conLog}(s,\,[((\mathrm{ajustarCaudal},\,cO-uCM),\;0)])\bigr)
    \end{aligned}
    & \text{si } est_{\mathrm{bolsa}} \in
      \{\mathrm{bolsaBaja},\,\mathrm{esperandoDetencion},\,\mathrm{vacia}\}
    \\[3ex]
    \begin{aligned}
      \bigl(&uCM,\;cO,\;0,\;\mathrm{normal},\;est_{\mathrm{bolsa}},\;\tau_{\mathrm{bolsa}}-e,\; \\
            &\mathrm{conLog}(s,\,[((\mathrm{ajustarCaudal},\,cO-uCM),\;0)])\bigr)
    \end{aligned}
    & \text{en otro caso}
  \end{cases}
\]

\paragraph{Rol 3 – Ciclo de bolsa.}
Transición externa — señal de fin de bolsa:
\[
  \delta_{\mathrm{ext}}(s,\,e,\,(\mathrm{finBolsa},2)) =
  \begin{cases}
    \begin{aligned}
      \bigl(&uCM,\,cO,\,tol,\,est_{\mathrm{flujo}},\,\mathrm{bolsaBaja},\,\tau_{\mathrm{bolsa}}-e,\, \\
            &\mathrm{conLog}(s,\,[(\mathrm{alarmaBaja},\;0)])\bigr)
    \end{aligned}
    & \text{si } est_{\mathrm{bolsa}} = \mathrm{llena}
    \\[3ex]
    \begin{aligned}
      \bigl(&uCM,\,cO,\,tol,\,est_{\mathrm{flujo}},\,est_{\mathrm{bolsa}},\;\tau_{\mathrm{bolsa}}-e,\; \\
            &[(a_1,\,\max(0,d_1-e)),\,(a_2,d_2),\ldots]\bigr)
    \end{aligned}
    & \begin{aligned}
        &\text{si } est_{\mathrm{bolsa}} \in \{\mathrm{bolsaBaja},\,\mathrm{esperandoDetencion},\,\mathrm{vacia}\} \\
        &\quad\land \mathit{acciones} = [(a_1,d_1),(a_2,d_2),\ldots]
      \end{aligned}
  \end{cases}
\]

Transiciones internas — ciclo de vida de la bolsa:
\begin{align*}
  \delta_{\mathrm{int}}(uCM,\,cO,\,tol,\,est_{\mathrm{flujo}},\,\mathrm{bolsaBaja},\,\tau_{\mathrm{bolsa}},\,[(\mathrm{alarmaBaja},0)])
  &= (uCM,\;cO,\;tol,\;est_{\mathrm{flujo}},\;\mathrm{esperandoDetencion},\;60,\;[])
\end{align*}

\begin{align*}
  \delta_{\mathrm{int}}(uCM,\,cO,\,tol,\,est_{\mathrm{flujo}},\,\mathrm{esperandoDetencion},\,0,\,[])
  &= (uCM,\;cO,\;tol,\;est_{\mathrm{flujo}},\;\mathrm{vacia},\;0,\;\mathrm{conLog}(s,\,[(\mathrm{detenerBomba},\;0)]))
\end{align*}

\begin{align*}
  \delta_{\mathrm{int}}(uCM,\,cO,\,tol,\,est_{\mathrm{flujo}},\,\mathrm{vacia},\,0,\,[(\mathrm{detenerBomba},0)])
  &= (uCM,\;cO,\;tol,\;est_{\mathrm{flujo}},\;\mathrm{vacia},\;\infty,\;[])
\end{align*}

\paragraph{Función de salida y transición interna general.}
\[
  \lambda(\ldots,\,[(a_1,d_1),(a_2,d_2),\ldots],\,\ldots) \;=\;
  \begin{cases}
    (a_1,\, 1) & \text{si } a_1 \in \{\mathrm{detenerBomba}\} \cup \{(\mathrm{ajustarCaudal},\,\Delta)\} \\
    (a_1,\, 0) & \text{si } a_1 \in \{\mathrm{alarmaBaja},\,\mathrm{alarmaMedia},\,\mathrm{alarmaCritica}\} \\
    (a_1,\, 9) & \text{si } a_1 = (\mathrm{registrarEvento},\, \cdot)
  \end{cases}
\]

\[
  \delta_{\mathrm{int}}(\ldots,\,[(a_1,d_1),(a_2,d_2),\ldots],\,\ldots)
  \;=\; (\ldots,\,[(a_2,d_2),\ldots],\,\ldots)
\]
